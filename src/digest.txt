import { Product } from "../domain/product"
import { ProductRepository } from "../domain/productRepository"

export class ProductRepositoryImpl implements ProductRepository {
  products: Product[] = [
    { id: 1, name: "Product 1", price: 100 },
    { id: 2, name: "Product 2", price: 200 }
  ]

  getProducts(): Product[] {
    return this.products
  }

  createProduct(product: Product): void {
    this.products.push(product)
  }

  updateProduct(product: Product): void {
    this.products = this.products.map(current => 
      current.id === product.id ? product : current
    )
  }
  
  deleteProduct(productId: number): void {
    this.products = this.products.filter(currente => 
      currente.id !== productId
    )
  }
}

import { View } from "react-native"
import { useState } from "react"
import { Product } from "../../domain/product"

interface ProductFormProps {
  onCreateProduct: (product: Product) => void
}

export const ProductForm = ({ onCreateProduct }: ProductFormProps) => {
  const [name, setName] = useState("")
  const [price, setPrice] = useState(0)

  const onSubmit: React.SubmitEventHandler<HTMLFormElement> = (e) => {
    e.preventDefault()

    const product: Product = {
      id: Date.now(),
      name: name,
      price: price
    }
    
    setName("")
    setPrice(0)
    
    onCreateProduct(product)
  }

  return (
    <View>
      <form onSubmit={onSubmit}>
        <input
          placeholder="Name"
          value={name}
          onChange={(e)=>setName(e.target.value)}
          />
        <input
          placeholder="Price"
          value={price}
          onChange={(e)=>setPrice(Number(e.target.value))}
          />
        <button type="submit">Add Product</button>
      </form>
    </View>
  )
}

import { useState, useEffect, use } from "react";
import { Product } from "../../domain/product";
import { ProductRepositoryImpl } from "../../data/productRepositoryImpl";

const respository = new ProductRepositoryImpl()

export function useProducts() {
  const [products, setProducts] = useState<Product[]>([]);

  const getProducts = () => {
    const products = respository.getProducts()
    setProducts(products)
  }

  const createProduct = (product: Product) => {
    respository.createProduct(product)
    setProducts((prev) => [...prev, product])
  }

  const updateProduct = (product: Product) => {
    respository.updateProduct(product)
    setProducts((prev) => prev.map((
      current => current.id === product.id ? product : current
    )))
  }

  const deleteProduct = (productId: number) => {
    respository.deleteProduct(productId)
    setProducts((prev) => prev.filter(
      current => current.id !== productId
    ))
  }

  useEffect(() => {
    getProducts()
  }, [])

  return {
    products,
    createProduct,
    updateProduct,
    deleteProduct
  }
}
